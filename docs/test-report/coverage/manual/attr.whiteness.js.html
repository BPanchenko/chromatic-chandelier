import { RadialDistance } from './coord.radius.js';
import createRange from './trunk/fn.init-range.js';
import round from './trunk/fn.round.js';
 
/**
 * Whiteness is usually measured using instruments such as spectrophotometers or colorimeters, which calculate Whiteness by measuring the reflectivity of a material to a specific light source. Whiteness measurement methods vary according to different industry standards. Common Whiteness measurement methods include:
 * 1. _Reflectance method_: Use a spectrophotometer or integrating sphere device to measure the reflectance of a material at a specific wavelength (such as 457 nm blue light). This method is commonly used in industries such as paper, plastics and textiles to evaluate the visual Whiteness of materials.
 * 1. _Integrating sphere method_: By placing the sample inside an integrating sphere, the reflected light intensity of the material under a D65 or C light source is measured to simulate the Whiteness effect under daylight conditions. This method is particularly suitable for materials containing fluorescent whitening agents.
 * 1. _Fluorescence Whiteness method_: For materials containing fluorescent whitening agents, the Whiteness enhancement effect of the material under ultraviolet light is measured. The fluorescence Whiteness method is applicable to the paper, textile and plastic industries to help evaluate the whitening effect.
 * 1. _CIE Whiteness Index Calculation_: According to ISO 11475 and ISO 11476 standards, the Whiteness index is calculated using the CIE formula, which is applicable to a variety of materials and is more consistent with the human eye's perception of Whiteness.
 *
 * [WIKI](https://en.wikipedia.org/wiki/Whiteness_(colorimetry))
 * | [Measurement Methods](http://www.3nh.com/en/news/161-991.html)
 *
 * @type {import('library').Dimension}
 */
const Whiteness = Object.freeze({
	$coord: RadialDistance,
	ident: 'whiteness',
	bytes: 8,
	domain: createRange(0, 100),
	input: val => round(val / 100),
	output: coord => round(coord * 100),
	short: 'W',
	precision: 2,
	unit: '%',
});
 
export { Whiteness as default, Whiteness };
